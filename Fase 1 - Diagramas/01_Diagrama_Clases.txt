package "Partida y progresion" {

    enum EtapaPartida {
        PREPARACION
        COMBATE
        ELECCION_MEJORA
        FINALIZADA
    }

    enum TipoMejora {
        EXPERIENCIA
        CREDITOS
    }

    class Partida {
        - equipo: Equipo
        - tienda: Tienda
        - batallaActual: Batalla
        - generador: GeneradorEncuentros
        - etapa: EtapaPartida
        - ciclo: int
        - estres: int
        - limiteEstres: int = 5
        - multiplicadorExperiencia: double
        - multiplicadorCreditos: double
        + iniciarNueva(): void
        + iniciarSiguienteCombate(): Batalla
        + procesarFinCombate(): void
        + elegirMejora(tipo: TipoMejora): boolean
        + comprar(oferta: Oferta): boolean
        + equipar(empleado: Empleado, objeto: Equipamiento): boolean
        + desequipar(empleado: Empleado, objeto: Equipamiento): boolean
        + usarConsumible(item: Consumible, objetivo: Empleado): boolean
        + puedeGuardar(): boolean
        + estaFinalizada(): boolean
    }

    class GeneradorEncuentros {
        + crearEnemigos(ciclo: int, esJefe: boolean): List<Enemigo>
    }

    class Recompensa {
        - experiencia: int
        - creditos: int
        + calcularExperiencia(multiplicador: double): int
        + calcularCreditos(multiplicador: double): int
    }

    class PersistenciaPartida {
        + guardar(partida: Partida, ruta: String): void
        + cargar(ruta: String): Partida
    }
}

    class TicketPendiente {

}
